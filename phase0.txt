-- PROFILES: one row per user, created automatically at sign-up
create table public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text,
  email text,
  created_at timestamptz not null default now()
);

create function public.handle_new_user()
returns trigger
language plpgsql
security definer set search_path = ''
as $$
begin
  insert into public.profiles (id, full_name, email)
  values (new.id, new.raw_user_meta_data->>'full_name', new.email);
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- PLANS: the 5 plans (the server's copy of the prices)
create table public.plans (
  id int primary key,
  name text not null,
  days int not null,
  amount int not null
);

-- SUBSCRIPTIONS: a user's memberships
create table public.subscriptions (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid() references auth.users(id) on delete cascade,
  plan_id int not null references public.plans(id),
  status text not null default 'pending'
    check (status in ('pending', 'active', 'cancelled', 'expired')),
  method text,
  start_date timestamptz,
  end_date timestamptz,
  created_at timestamptz not null default now(),
  cancelled_at timestamptz
);

-- a user can have only ONE pending or active membership at a time
create unique index one_open_subscription_per_user
  on public.subscriptions (user_id)
  where status in ('pending', 'active');

-- PAYMENTS: powers the Payment History screen
create table public.payments (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid() references auth.users(id) on delete cascade,
  subscription_id uuid references public.subscriptions(id) on delete set null,
  plan_id int not null references public.plans(id),
  amount int not null,
  method text not null,
  paid_at timestamptz not null default now()
);

-- ROW LEVEL SECURITY: this is what makes the public key safe
alter table public.profiles      enable row level security;
alter table public.plans         enable row level security;
alter table public.subscriptions enable row level security;
alter table public.payments      enable row level security;

create policy "plans are readable by everyone"
  on public.plans for select using (true);

create policy "read own profile"
  on public.profiles for select to authenticated
  using (id = (select auth.uid()));
create policy "update own profile"
  on public.profiles for update to authenticated
  using (id = (select auth.uid())) with check (id = (select auth.uid()));

create policy "read own subscriptions"
  on public.subscriptions for select to authenticated
  using (user_id = (select auth.uid()));
create policy "create own subscriptions"
  on public.subscriptions for insert to authenticated
  with check (user_id = (select auth.uid()));
create policy "update own subscriptions"
  on public.subscriptions for update to authenticated
  using (user_id = (select auth.uid())) with check (user_id = (select auth.uid()));

create policy "read own payments"
  on public.payments for select to authenticated
  using (user_id = (select auth.uid()));
create policy "create own payments"
  on public.payments for insert to authenticated
  with check (user_id = (select auth.uid()));
